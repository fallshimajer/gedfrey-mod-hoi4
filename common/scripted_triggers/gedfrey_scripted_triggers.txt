	OR = {
		tag = USA

	}
}

gedfrey_is_minor = {
	NOT = { gedfrey_is_major = yes }
	NOT = { gedfrey_is_medium = yes }
}

# ============================================================
# Factory Count Triggers (civilian + military combined)
# Uses > instead of >= for HOI4 integer trigger compatibility.
# "> 29" is equivalent to ">= 30".
#
# Industry thresholds:
#   N1: minor>=30, medium>=50, major>=70
#   N2: minor>=60, medium>=100, major>=140
#   N3: minor>=90, medium>=150, major>=210
# ============================================================

gedfrey_factories_minor_n1 = {       # civ + mil >= 30
	OR = {
		num_of_civilian_factories > 29
		AND = { num_of_civilian_factories > 24  num_of_military_factories > 4  }
		AND = { num_of_civilian_factories > 19  num_of_military_factories > 9  }
		AND = { num_of_civilian_factories > 14  num_of_military_factories > 14 }
		AND = { num_of_civilian_factories > 9   num_of_military_factories > 19 }
		AND = { num_of_civilian_factories > 4   num_of_military_factories > 24 }
		num_of_military_factories > 29
	}
}
