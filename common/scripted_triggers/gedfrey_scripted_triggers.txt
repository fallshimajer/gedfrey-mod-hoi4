# --- Country Classification ---

gedfrey_is_major = {
	OR = {
		tag = USA
		tag = SOV
		tag = GER
		tag = ENG
		tag = JAP
	}
}

gedfrey_is_medium = {
	OR = {
		tag = FRA
		tag = ITA
		tag = ROC
		tag = CHI
		tag = BRA
		tag = RAJ
		tag = POL
		tag = TUR
		tag = SPR
	}

gedfrey_is_minor = {
	NOT = { gedfrey_is_major = yes }
	NOT = { gedfrey_is_medium = yes }
}

gedfrey_is_naval_power = {
	OR = {
		tag = USA
		tag = ENG
	}
}

# ============================================================
# Factory Count Triggers (civilian + military combined)
#
# HOI4 cannot add two variables in a trigger, so each threshold
# is expressed as an OR of enumerated (civ, mil) pairs at step 5/10/20.
# Maximum imprecision: ~step size factories.
#
# Naming convention: gedfrey_factories_[tier]_[level]
#   tier  = minor | medium | major
#   level = n1 | n2 | n3
#
# Industry thresholds:
#   N1: minor>=30, medium>=50, major>=70
#   N2: minor>=60, medium>=100, major>=140
#   N3: minor>=90, medium>=150, major>=210
# ============================================================

gedfrey_factories_minor_n1 = {       # civ + mil >= 30  (step 5)
	OR = {
		num_of_civilian_factories >= 30
		AND = { num_of_civilian_factories >= 25  num_of_military_factories >= 5  }
		AND = { num_of_civilian_factories >= 20  num_of_military_factories >= 10 }
		AND = { num_of_civilian_factories >= 15  num_of_military_factories >= 15 }
		AND = { num_of_civilian_factories >= 10  num_of_military_factories >= 20 }
		AND = { num_of_civilian_factories >= 5   num_of_military_factories >= 25 }
		num_of_military_factories >= 30
	}
}
